# Power BI writes PBIP text files with CRLF line endings and rewrites them
# every time Desktop opens the project. If git normalises those endings you get
# spurious whole-file diffs on every open. Pinning them to CRLF stops the churn
# regardless of each contributor's core.autocrlf setting.

* text=auto eol=lf

*.tmdl      text eol=crlf
*.pbip      text eol=crlf
*.pbism     text eol=crlf
*.platform  text eol=crlf
*.json      text eol=crlf

# data and binaries - never EOL-converted, never diffed
*.csv    binary
*.abf    binary
*.pbix   binary
*.pbit   binary
*.zip    binary
*.png    binary
